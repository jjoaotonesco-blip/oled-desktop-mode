cmake_minimum_required(VERSION 3.20)
project(OledDesktopMode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(state_tests tests/state_tests.cpp)
target_include_directories(state_tests PRIVATE src)
add_test(NAME state_safety COMMAND state_tests)
if(WIN32)
  enable_language(RC)
  set(ADLX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/ADLX" CACHE PATH "Official AMD ADLX SDK")
  if(NOT EXISTS "${ADLX_ROOT}/SDK/Include/IDisplaySettings.h")
    message(FATAL_ERROR "Clone GPUOpen-LibrariesAndSDKs/ADLX into external/ADLX; see README.md")
  endif()
  add_executable(OledDesktopMode src/main.cpp src/oled.rc
    "${ADLX_ROOT}/SDK/ADLXHelper/Windows/Cpp/ADLXHelper.cpp"
    "${ADLX_ROOT}/SDK/Platform/Windows/WinAPIs.cpp")
  target_include_directories(OledDesktopMode PRIVATE "${ADLX_ROOT}")
  target_compile_definitions(OledDesktopMode PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(OledDesktopMode PRIVATE dwmapi shell32 wtsapi32 user32 advapi32)
  if(MINGW)
    target_compile_options(OledDesktopMode PRIVATE -Wno-unknown-attributes)
    target_link_options(OledDesktopMode PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
